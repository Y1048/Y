cmake_minimum_required(VERSION 3.16)
project(g1_velocity_mink_keypad_right_arm LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
if(NOT DEFINED UNITREE_SDK2_ROOT)
  set(UNITREE_SDK2_ROOT "$ENV{HOME}/unitree_sdk2-main")
endif()
if(NOT DEFINED TORCH_PYTHON_ROOT)
  execute_process(COMMAND python3 -c "import pathlib, torch; print(pathlib.Path(torch.__file__).parent)"
    OUTPUT_VARIABLE TORCH_PYTHON_ROOT OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE TORCH_RESULT)
  if(NOT TORCH_RESULT EQUAL 0)
    message(FATAL_ERROR "Could not discover Python torch")
  endif()
endif()
if(NOT DEFINED TORCH_CXX11_ABI)
  execute_process(COMMAND python3 -c "import torch; print(1 if torch.compiled_with_cxx11_abi() else 0)"
    OUTPUT_VARIABLE TORCH_CXX11_ABI OUTPUT_STRIP_TRAILING_WHITESPACE)
endif()
set(UNITREE_ARCH_LIB "${UNITREE_SDK2_ROOT}/lib/${CMAKE_SYSTEM_PROCESSOR}")
set(DDS_INCLUDE_DIRS /usr/local/include/ddscxx)
set(DDS_LIBRARY_DIRS /usr/local/lib)
if(EXISTS "${UNITREE_SDK2_ROOT}/thirdparty/include/ddsc")
  list(APPEND DDS_INCLUDE_DIRS "${UNITREE_SDK2_ROOT}/thirdparty/include" "${UNITREE_SDK2_ROOT}/thirdparty/include/ddscxx")
  list(APPEND DDS_LIBRARY_DIRS "${UNITREE_SDK2_ROOT}/thirdparty/lib/${CMAKE_SYSTEM_PROCESSOR}")
endif()
find_package(Threads REQUIRED)
enable_testing()
add_executable(test_leg_policy_switch test_leg_policy_switch.cpp)
target_compile_options(test_leg_policy_switch PRIVATE -Wall -Wextra -Wpedantic -Werror=return-type)
add_test(NAME leg_policy_switch COMMAND test_leg_policy_switch)
add_executable(test_velocity_keypad_contract test_velocity_keypad_contract.cpp)
target_compile_options(test_velocity_keypad_contract PRIVATE -Wall -Wextra -Wpedantic -Werror=return-type)
add_test(NAME velocity_keypad_contract COMMAND test_velocity_keypad_contract)
add_executable(test_initial_ready_continuous_gait test_initial_ready_continuous_gait.cpp)
target_compile_options(test_initial_ready_continuous_gait PRIVATE -Wall -Wextra -Wpedantic -Werror=return-type)
add_test(NAME initial_ready_continuous_gait COMMAND test_initial_ready_continuous_gait)
add_executable(g1_velocity_mink_keypad_right_arm g1_velocity_mink_keypad_right_arm.cpp)
target_include_directories(g1_velocity_mink_keypad_right_arm PRIVATE
  "${UNITREE_SDK2_ROOT}/include" ${DDS_INCLUDE_DIRS}
  "${TORCH_PYTHON_ROOT}/include" "${TORCH_PYTHON_ROOT}/include/torch/csrc/api/include")
target_link_directories(g1_velocity_mink_keypad_right_arm PRIVATE "${UNITREE_ARCH_LIB}" ${DDS_LIBRARY_DIRS})
target_link_libraries(g1_velocity_mink_keypad_right_arm PRIVATE unitree_sdk2 ddsc ddscxx
  "${TORCH_PYTHON_ROOT}/lib/libtorch.so" "${TORCH_PYTHON_ROOT}/lib/libtorch_cpu.so"
  "${TORCH_PYTHON_ROOT}/lib/libc10.so" Threads::Threads)
target_compile_definitions(g1_velocity_mink_keypad_right_arm PRIVATE
  _GLIBCXX_USE_CXX11_ABI=${TORCH_CXX11_ABI} G1_VELOCITY_POLICY=1)
target_compile_options(g1_velocity_mink_keypad_right_arm PRIVATE -Wall -Wextra -Wpedantic -Werror=return-type)
set_target_properties(g1_velocity_mink_keypad_right_arm PROPERTIES
  BUILD_RPATH "${TORCH_PYTHON_ROOT}/lib" INSTALL_RPATH "${TORCH_PYTHON_ROOT}/lib")
